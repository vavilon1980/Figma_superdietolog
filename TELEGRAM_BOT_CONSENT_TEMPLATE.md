## Реализация в коде бота

### Python (aiogram) пример:

```python
from aiogram import types
from aiogram.types import InlineKeyboardMarkup, InlineKeyboardButton

# Клавиатура согласия
def get_consent_keyboard():
    keyboard = InlineKeyboardMarkup(row_width=1)
    keyboard.add(
        InlineKeyboardButton("✅ Согласен и продолжаю", callback_data="consent_yes"),
        InlineKeyboardButton("📄 Политика конфиденциальности", url="https://superdietolog.ru/#privacy"),
        InlineKeyboardButton("❌ Отмена", callback_data="consent_no")
    )
    return keyboard

# Обработчик /start
@dp.message_handler(commands=['start'])
async def cmd_start(message: types.Message):
    # Проверяем, давал ли пользователь согласие
    user_consent = await db.get_user_consent(message.from_user.id)
    
    if not user_consent:
        await message.answer(
            "🤖 Добро пожаловать в AI-Супердиетолог!\n\n"
            "⚠️ ВАЖНАЯ ИНФОРМАЦИЯ О ПЕРСОНАЛЬНЫХ ДАННЫХ\n\n"
            "Для работы я буду обрабатывать:\n"
            "• Ваш Telegram ID\n"
            "• Ответы на вопросы о здоровье\n"
            "• Историю переписки\n\n"
            "📄 Подробнее: https://superdietolog.ru/#privacy\n\n"
            "Продолжая, вы соглашаетесь с условиями обработки данных.",
            reply_markup=get_consent_keyboard()
        )
    else:
        # Пользователь уже давал согласие
        await show_main_menu(message)

# Обработчик согласия
@dp.callback_query_handler(text="consent_yes")
async def consent_given(call: types.CallbackQuery):
    # Сохраняем согласие в БД
    await db.save_user_consent(
        user_id=call.from_user.id,
        consent=True,
        consent_date=datetime.now()
    )
    
    await call.message.edit_text(
        "✅ Спасибо! Ваше согласие зафиксировано.\n\n"
        "Теперь давайте начнем работу!"
    )
    await show_main_menu(call.message)

# Команда /delete_data
@dp.message_handler(commands=['delete_data'])
async def cmd_delete_data(message: types.Message):
    keyboard = InlineKeyboardMarkup(row_width=2)
    keyboard.add(
        InlineKeyboardButton("✅ Да, удалить", callback_data="confirm_delete"),
        InlineKeyboardButton("❌ Нет, оставить", callback_data="cancel_delete")
    )
    
    await message.answer(
        "⚠️ УДАЛЕНИЕ ПЕРСОНАЛЬНЫХ ДАННЫХ\n\n"
        "Вы действительно хотите удалить все свои данные?\n\n"
        "Будет удалено:\n"
        "• История переписки\n"
        "• Результаты анализа\n"
        "• Персональные рекомендации\n\n"
        "⚠️ Это действие нельзя отменить!",
        reply_markup=keyboard
    )

@dp.callback_query_handler(text="confirm_delete")
async def confirm_delete(call: types.CallbackQuery):
    # Удаляем все данные пользователя
    await db.delete_user_data(call.from_user.id)
    
    await call.message.edit_text(
        "✅ Все ваши данные успешно удалены.\n\n"
        "Если захотите вернуться, просто напишите /start"
    )
```
